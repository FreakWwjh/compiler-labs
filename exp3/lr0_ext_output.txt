State 2:
  F -> . ( E )

  E -> E . + T

  T -> T . * F

State 6:

State 7:

  F -> ( E . )

State 8:
  F -> - F .

State 9:
  E -> E + . T
  T -> . T * F
  T -> . F
  F -> . ( E )
  F -> . - F
  F -> . id

State 10:
  T -> T * . F
  F -> . ( E )
  F -> . - F
  F -> . id
Kernel: T -> T * . F

State 11:
  F -> ( E ) .
Kernel: F -> ( E ) .

State 12:
  E -> E + T .
  T -> T . * F
Kernel: E -> E + T ., T -> T . * F

State 13:
  T -> T * F .
Kernel: T -> T * F .
