==================================================
0: E' -> E
1: E -> E + T
2: E -> T
3: T -> T * F
4: T -> F
5: F -> ( E )
6: F -> id

LR(0) 项目集规范族
==================================================
State 0:
  E' -> . E
  E -> . E + T
  E -> . T
  T -> . T * F
  T -> . F
  F -> . ( E )
  F -> . id
Kernel: E' -> . E

State 1:
  E -> . E + T
  E -> . T
  T -> . T * F
  T -> . F
  F -> . ( E )
  F -> ( . E )
  F -> . id
Kernel: F -> ( . E )

State 2:
  E' -> E .
  E -> E . + T
Kernel: E' -> E ., E -> E . + T

State 3:
  T -> F .
Kernel: T -> F .

State 4:
  E -> T .
  T -> T . * F
Kernel: E -> T ., T -> T . * F

State 5:
  F -> id .
Kernel: F -> id .

State 6:
  E -> E . + T
  F -> ( E . )
Kernel: E -> E . + T, F -> ( E . )

State 7:
  E -> E + . T
  T -> . T * F
  T -> . F
  F -> . ( E )
  F -> . id
Kernel: E -> E + . T

State 8:
  T -> T * . F
  F -> . ( E )
  F -> . id
Kernel: T -> T * . F

State 9:
  F -> ( E ) .
Kernel: F -> ( E ) .

State 10:
  E -> E + T .
  T -> T . * F
Kernel: E -> E + T ., T -> T . * F

State 11:
  T -> T * F .
Kernel: T -> T * F .
