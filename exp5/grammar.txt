Prog -> DeclList
DeclList -> DeclList SEMI Decl | Decl | ε
Decl -> FunDecl | Stmt
FunDecl -> FunDeclHead LBR StmtList RBR
FunDeclHead -> Type ID LPAR RPAR
Type -> INT | FLOAT | VOID
StmtList -> StmtList SEMI Stmt | Stmt | ε
Stmt -> VarDecl | Assign | Print | Return | IfStmt | WhileStmt | FunCallStmt
VarDecl -> Type ID | Type ID ASG Expr
Assign -> ID ASG Expr
Print -> PRINT Expr
Return -> RETURN Expr | RETURN
IfStmt -> IF LPAR Expr RPAR LBR StmtList RBR
WhileStmt -> WHILE LPAR Expr RPAR LBR StmtList RBR
FunCallStmt -> ID LPAR RPAR
Expr -> Expr ADD Term | Expr SUB Term | Expr LT Term | Expr LE Term | Expr EQ Term | Term
Term -> Term MUL Fact | Fact
Fact -> ID | INT_NUM | FLOAT_NUM | LPAR Expr RPAR | ID LPAR RPAR
